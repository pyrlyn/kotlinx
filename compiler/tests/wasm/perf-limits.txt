# Performance checks for compiler/tests/wasm/perf.sh: <name> <bench case> <a> <b> <limit>; fails when
# median(a) / median(b) > limit. a, b: a build (default = newkotlinc defaults, malloc = --alloc malloc with the
# wasi-libc malloc archive), optionally :<node flags> with ',' for spaces. Limits leave room for a loaded machine;
# the ratio measured when each check was added is in the comment above it.
# --alloc pool (wasm default) against wasi-libc malloc: 0.50x on bigheap, 0.82x on alloc (Apple M3 Max, node 26).
pool-bigheap bigheap default malloc 0.75
# pool-alloc: 0.79-1.03x on the hosted macos-15 runner, so its limit only catches a large regression.
pool-alloc alloc default malloc 1.20
# Tier-up: nbody's advance, called from a loop of main, must run as optimized code. Default node (Liftoff, then
# TurboFan on a later call) against --no-liftoff: 0.79x with the call kept out of main, 1.72x when it was inlined.
tierup-nbody nbody default default:--no-liftoff 1.20
# Cycle collection cost: perceus (default) against --gc rc, which never collects cycles (and runs slower from the
# leaked memory). Trial deletion over class layouts and the recorded gray set (native/gc/zig/src/common/
# trial_deletion.zig): cycles 0.83x -> 0.50x, bigheap 0.92x -> 0.67x.
cc-cycles cycles default rc 0.70
cc-bigheap bigheap default rc 0.82
